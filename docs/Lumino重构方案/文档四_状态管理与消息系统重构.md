## 二、重构方案：事件驱动架构

### 2.1 核心思想

**用不可变事件流替代直接状态修改**。

所有状态变更统一走 `EditorSession::apply(event)`，由框架自动推导：
- 文档更新
- 历史记录
- 渲染增量
- 音频增量

### 2.2 新状态结构

```rust
// lumino-core/src/document.rs

    pub tracks: Vec<Track>,

    pub time_signatures: Vec<TimeSignature>,

    /// 唯一可变入口
    pub fn apply(&mut self, event: &DocumentEvent) -> Result<(), DocumentError> {
        match event {
            DocumentEvent::NotesInserted { track, range } => {
                // 验证并插入
            }
            DocumentEvent::NotesDeleted { track, range } => {
                // 验证并删除
            }
            DocumentEvent::NotesMoved { track, delta_tick, delta_key } => {
                // 验证并移动
            }
            // ...
        }
        Ok(())
    }

// lumino-editor/src/view.rs

/// 纯视图状态（不可序列化，随时可重建）
pub struct EditorView {
    pub scroll: (f32, f32),
    pub zoom: (f32, f32),
    pub selection: Selection,
    pub current_tool: Tool,
    pub hovered_note: Option<usize>,
    pub canvas_size: (f32, f32),
}

impl EditorView {
    /// 根据事件更新视图（如选择框变化时更新 selection）
    pub fn apply(&mut self, event: &DocumentEvent) {
        match event {
            DocumentEvent::SelectionChanged { track, selected } => {
                self.selection = Selection::new(*track, selected.clone());
            }
            _ => {}
        }
    }

// lumino-editor/src/session.rs

/// 编辑器会话：文档 + 视图 + 历史 + 增量队列
pub struct EditorSession {
    document: Document,
    view: EditorView,
    history: History,
    gfx_delta: GfxDeltaQueue,
    audio_delta: AudioDeltaQueue,
}

impl EditorSession {
    /// 唯一的状态变更入口
    pub fn apply(&mut self, event: DocumentEvent) -> Result<(), EditError> {
        // 1. 应用到文档（验证）
        self.document.apply(&event)?;

        // 2. 更新视图
        self.view.apply(&event);

        // 3. 记录历史（自动推导反向操作）
        if let Some(inverse) = event.inverse() {
            self.history.push(event, inverse);
        }

        // 4. 推导渲染增量
        self.gfx_delta.push(&self.document, &event);

        // 5. 推导音频增量
        self.audio_delta.push(&self.document, &event);

        Ok(())
    }

    /// 撤销
    pub fn undo(&mut self) -> Result<(), EditError> {
        if let Some(inverse) = self.history.pop() {
            self.apply(inverse)?;
        }
        Ok(())
    }

    /// 获取当前音轨音符（简化访问）
    pub fn current_track_notes(&self) -> &[NoteEvent] {
        self.document.tracks[self.view.current_track()].notes()
    }
}
```

### 2.3 事件定义

```rust
// lumino-core/src/events.rs

/// 文档级别事件（不可变，可序列化）
#[derive(Debug, Clone, PartialEq)]
pub enum DocumentEvent {
    // === 音符操作 ===
    NotesInserted {
        track: usize,
        notes: Vec<NoteEvent>,
    },
    NotesDeleted {
        track: usize,
        indices: Vec<usize>,
    },
    NotesMoved {
        track: usize,
        indices: Vec<usize>,
        delta_tick: i32,
        delta_key: i16,
    },
    NotesResized {
        track: usize,
        indices: Vec<usize>,
        delta_duration: i32,
    },
    NoteVelocityChanged {
        track: usize,
        indices: Vec<usize>,
        new_velocity: u8,
    },

    // === 轨道操作 ===
    TrackAdded {
        index: usize,
        name: String,
    },
    TrackRemoved {
        index: usize,
    },
    TrackReordered {
        old_index: usize,
        new_index: usize,
    },
    TrackRenamed {
        index: usize,
        new_name: String,
    },
    TrackMuted {
        index: usize,
        muted: bool,
    },

    // === 全局事件 ===
    TempoChanged {
        tick: u32,
        new_tempo: u32,
    },
    TimeSignatureChanged {
        tick: u32,
        numerator: u8,
        denominator: u8,
    },
    KeySignatureChanged {
        tick: u32,
        key: Key,
        mode: Mode,
    },

    // === 选择操作 ===
    SelectionChanged {
        track: usize,
        selected: Vec<usize>,
    },
    SelectionCleared,

    // === 视图操作（不影响文档，只影响视图）===
    ViewScrolled {
        scroll: (f32, f32),
    },
    ViewZoomed {
        zoom: (f32, f32),
        anchor: (f32, f32),
    },
    ToolChanged {
        tool: Tool,
    },
}

impl DocumentEvent {
    /// 推导反向事件（用于撤销）
    pub fn inverse(&self) -> Option<DocumentEvent> {
        match self {
            DocumentEvent::NotesInserted { track, notes } => {
                // 计算插入后的索引，生成删除事件
                Some(DocumentEvent::NotesDeleted {
                    track: *track,
                    indices: (0..notes.len()).collect(),
                })
            }
            DocumentEvent::NotesDeleted { track, indices } => {
                // 需要保存被删除的音符数据（History 中存储）
                None // 由 History 处理
            }
            DocumentEvent::NotesMoved { track, indices, delta_tick, delta_key } => {
                Some(DocumentEvent::NotesMoved {
                    track: *track,
                    indices: indices.clone(),
                    delta_tick: -delta_tick,
                    delta_key: -delta_key,
                })
            }
            DocumentEvent::ViewScrolled { .. }
            | DocumentEvent::ViewZoomed { .. }
            | DocumentEvent::ToolChanged { .. } => None, // 视图操作不记录历史
            _ => None,
        }
    }

    /// 是否影响文档（需要保存/同步）
    pub fn affects_document(&self) -> bool {
        !matches!(self,
            DocumentEvent::ViewScrolled { .. }
            | DocumentEvent::ViewZoomed { .. }
            | DocumentEvent::ToolChanged { .. }
            | DocumentEvent::SelectionChanged { .. }
            | DocumentEvent::SelectionCleared
        )
    }
}
```

### 2.4 历史记录重构

```rust
// lumino-editor/src/history.rs

/// 历史条目
struct HistoryEntry {
    event: DocumentEvent,      // 正向事件
    inverse: DocumentEvent,    // 反向事件
    timestamp: Instant,
}

/// 撤销/重做栈
pub struct History {
    undo_stack: Vec<HistoryEntry>,
    redo_stack: Vec<HistoryEntry>,
    max_size: usize,
}

impl History {
    pub fn push(&mut self, event: DocumentEvent, inverse: DocumentEvent) {
        self.undo_stack.push(HistoryEntry {
            event,
            inverse,
            timestamp: Instant::now(),
        });
        self.redo_stack.clear(); // 新操作清空 redo 栈

        // 限制栈大小
        if self.undo_stack.len() > self.max_size {
            self.undo_stack.remove(0);
        }
    }

    pub fn undo(&mut self) -> Option<DocumentEvent> {
        self.undo_stack.pop().map(|entry| {
            self.redo_stack.push(HistoryEntry {
                event: entry.inverse.clone(),
                inverse: entry.event,
                timestamp: entry.timestamp,
            });
            entry.inverse
        })
    }

    pub fn redo(&mut self) -> Option<DocumentEvent> {
        self.redo_stack.pop().map(|entry| {
            self.undo_stack.push(HistoryEntry {
                event: entry.inverse.clone(),
                inverse: entry.event,
                timestamp: entry.timestamp,
            });
            entry.inverse
        })
    }
}
```

### 2.5 渲染增量自动推导

```rust
// lumino-gfx/src/delta.rs

/// 渲染增量队列
pub struct GfxDeltaQueue {
    deltas: Vec<GfxDelta>,
}

#[derive(Debug, Clone)]
pub enum GfxDelta {
    /// 全量重建（切轨、加载、undo/redo）
    FullRebuild { track: usize },
    /// 增量更新（拖动等高频操作）
    PartialUpdate {
        track: usize,
        range: Range<usize>,
    },
    /// 视图变化（滚动/缩放，无需重建顶点）
    ViewChanged,
}

impl GfxDeltaQueue {
    pub fn push(&mut self, document: &Document, event: &DocumentEvent) {
        let delta = match event {
            DocumentEvent::NotesInserted { track, .. }
            | DocumentEvent::NotesDeleted { track, .. }
            | DocumentEvent::NotesMoved { track, .. }
            | DocumentEvent::NotesResized { track, .. } => {
                GfxDelta::PartialUpdate {
                    track: *track,
                    range: 0..document.tracks[*track].note_count(),
                }
            }
            DocumentEvent::TrackAdded { .. }
            | DocumentEvent::TrackRemoved { .. }
            | DocumentEvent::TrackReordered { .. }
            | DocumentEvent::TempoChanged { .. } => {
                GfxDelta::FullRebuild { track: 0 }
            }
            DocumentEvent::ViewScrolled { .. }
            | DocumentEvent::ViewZoomed { .. } => {
                GfxDelta::ViewChanged
            }
            _ => return, // 不生成增量
        };
        self.deltas.push(delta);
    }

    pub fn drain(&mut self) -> Vec<GfxDelta> {
        std::mem::take(&mut self.deltas)
    }
}
```

### 3.2 Action Handler Trait

```rust
// lumino-core/src/actions.rs

/// 动作处理器
/// 
/// 各模块实现此 trait 处理自己关心的动作。
pub trait ActionHandler: Send {
    /// 处理动作，返回是否消费
    fn handle(&mut self, action: &AppAction) -> Result<ActionResult, ActionError>;

    /// 声明自己关心的动作类型（用于优化分发）
    fn interests(&self) -> &'static [AppActionKind] {
        &[] // 默认关心所有
    }
}

#[derive(Debug, Clone, PartialEq)]
pub enum ActionResult {
    Consumed,           // 已处理
    Ignored,            // 不处理
    Propagate(AppAction), // 转换为新动作继续分发
}

/// 动作分发器
pub struct ActionDispatcher {
    handlers: Vec<Box<dyn ActionHandler>>,
}

impl ActionDispatcher {
    pub fn dispatch(&mut self, action: AppAction) -> Result<(), ActionError> {
        for handler in &mut self.handlers {
            match handler.handle(&action)? {
                ActionResult::Consumed => return Ok(()),
                ActionResult::Ignored => continue,
                ActionResult::Propagate(new_action) => {
                    return self.dispatch(new_action);
                }
            }
        }
        Ok(())
    }
}
```

### 3.3 Editor 实现 ActionHandler

```rust
// lumino-editor/src/lib.rs

impl ActionHandler for EditorSession {
    fn handle(&mut self, action: &AppAction) -> Result<ActionResult, ActionError> {
        match action {
            // 编辑操作 → 转换为 DocumentEvent
            AppAction::InsertNotes { track, notes } => {
                self.apply(DocumentEvent::NotesInserted {
                    track: *track,
                    notes: notes.clone(),
                })?;
                Ok(ActionResult::Consumed)
            }
            AppAction::DeleteSelectedNotes => {
                let selected = self.view.selection.indices().to_vec();
                self.apply(DocumentEvent::NotesDeleted {
                    track: self.view.current_track(),
                    indices: selected,
                })?;
                Ok(ActionResult::Consumed)
            }
            AppAction::Undo => {
                self.undo()?;
                Ok(ActionResult::Consumed)
            }
            AppAction::Redo => {
                self.redo()?;
                Ok(ActionResult::Consumed)
            }

            // 视图操作 → 直接更新视图
            AppAction::SetTool(tool) => {
                self.view.current_tool = *tool;
                Ok(ActionResult::Consumed)
            }
            AppAction::ZoomIn => {
                self.view.zoom.0 *= 1.1;
                Ok(ActionResult::Consumed)
            }

            // 不关心的动作
            _ => Ok(ActionResult::Ignored),
        }
    }

    fn interests(&self) -> &'static [AppActionKind] {
        &[
            AppActionKind::Edit,
            AppActionKind::View,
            AppActionKind::Track,
        ]
    }
}
```

```rust
// lumino-editor/src/session.rs

pub struct EditorSession {
    // ... 其他字段
    changed: bool, // 标记是否有变更
}

impl EditorSession {
    pub fn apply(&mut self, event: DocumentEvent) -> Result<(), EditError> {
        // ... 原有逻辑

        self.changed = true;
        Ok(())
    }

    /// 由 App 每帧调用，检查是否需要重绘
    pub fn take_changed(&mut self) -> bool {
        std::mem::replace(&mut self.changed, false)
    }
}

// lumino-app/src/app.rs
impl App {
    fn on_update(&mut self) {
        // 检查编辑器是否有变更
        if self.editor.take_changed() {
            self.window.request_redraw();
        }

        // 检查渲染增量
        let deltas = self.editor.gfx_delta.drain();
        self.gfx.apply_deltas(deltas);
    }
}
```
