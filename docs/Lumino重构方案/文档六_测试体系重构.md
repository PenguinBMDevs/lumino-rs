## 二、重构方案

### 2.1 测试分层体系

```
测试金字塔

        ┌─────────┐
        │  E2E    │  ← 少量，覆盖完整工作流
        │  (5%)   │     如：加载 MIDI → 编辑 → 导出
        ├─────────┤
        │ 集成测试 │  ← 中量，覆盖 crate 间交互
        │  (15%)  │     如：Editor + Gfx + Audio
        ├─────────┤
        │ 单元测试 │  ← 大量，覆盖核心逻辑
        │  (80%)  │     如：ChunkedList、Document、History
        └─────────┘
```

#### 测试命名

```rust
#[cfg(test)]
mod tests {
    use super::*;

    // 命名规范：test_<被测函数>_<场景>_<预期>

    #[test]
    fn test_insert_note_at_valid_position_inserts_correctly() {
        let mut doc = Document::new();
        let note = NoteEvent::new(100, 60, 100, 480);
        doc.insert_note(0, note).unwrap();

        assert_eq!(doc.track_notes(0).len(), 1);
        assert_eq!(doc.track_notes(0)[0].tick, 100);
    }

    #[test]
    fn test_insert_note_at_invalid_track_returns_error() {
        let mut doc = Document::new();
        let note = NoteEvent::new(100, 60, 100, 480);

        let result = doc.insert_note(999, note);
        assert!(matches!(result, Err(EditError::TrackNotFound(999))));
    }

    #[test]
    fn test_undo_insert_note_removes_note() {
        let mut doc = Document::new();
        let note = NoteEvent::new(100, 60, 100, 480);
        doc.insert_note(0, note).unwrap();

        doc.undo().unwrap();

        assert_eq!(doc.track_notes(0).len(), 0);
    }
}
```

#### 测试数据工厂

```rust
// lumino-tests/src/factories.rs

pub struct NoteFactory;

impl NoteFactory {
    pub fn c4(tick: u32, duration: u32) -> NoteEvent {
        NoteEvent::new(tick, 60, 100, duration)
    }

    pub fn d4(tick: u32, duration: u32) -> NoteEvent {
        NoteEvent::new(tick, 62, 100, duration)
    }

    pub fn e4(tick: u32, duration: u32) -> NoteEvent {
        NoteEvent::new(tick, 64, 100, duration)
    }

    pub fn chord(tick: u32, duration: u32) -> Vec<NoteEvent> {
        vec![
            Self::c4(tick, duration),
            Self::e4(tick, duration),
            NoteEvent::new(tick, 67, 100, duration), // G4
        ]
    }
}

pub struct DocumentFactory;

impl DocumentFactory {
    pub fn empty() -> Document {
        Document::new()
    }

    pub fn with_single_note() -> Document {
        let mut doc = Document::new();
        doc.insert_note(0, NoteFactory::c4(0, 480)).unwrap();
        doc
    }

    pub fn with_c_major_scale() -> Document {
        let mut doc = Document::new();
        let notes = vec![60, 62, 64, 65, 67, 69, 71, 72];
        for (i, key) in notes.iter().enumerate() {
            doc.insert_note(0, NoteEvent::new(i as u32 * 480, *key, 100, 480)).unwrap();
        }
        doc
    }
}
```

```rust
// crates/lumino-tests/src/editor_integration.rs

#[test]
fn test_load_midi_and_playback() {
    // 1. 创建 EditorSession
    let mut editor = EditorSession::new();

    // 2. 加载 MIDI
    let midi_data = include_bytes!("fixtures/c_major.mid");
    let document = Document::from_midi_bytes(midi_data).unwrap();
    editor.load_document(document);

    // 3. 验证加载成功
    assert_eq!(editor.document().track_count(), 1);
    assert_eq!(editor.document().track_notes(0).len(), 8);

    // 4. 播放
    editor.play().unwrap();
    assert!(editor.is_playing());

    // 5. 停止
    editor.stop();
    assert!(!editor.is_playing());
}

#[test]
fn test_edit_and_undo() {
    let mut editor = EditorSession::new();
    editor.load_document(DocumentFactory::with_single_note());

    // 删除音符
    editor.select_all();
    editor.delete_selected().unwrap();
    assert_eq!(editor.document().track_notes(0).len(), 0);

    // 撤销
    editor.undo().unwrap();
    assert_eq!(editor.document().track_notes(0).len(), 1);
}
```

```rust
#[cfg(test)]
mod tests {
    use super::*;

    // 使用 wgpu 的测试适配器
    async fn create_test_device() -> (wgpu::Device, wgpu::Queue) {
        let instance = wgpu::Instance::new(wgpu::InstanceDescriptor::default());
        let adapter = instance
            .request_adapter(&wgpu::RequestAdapterOptions::default())
            .await
            .unwrap();
        adapter
            .request_device(&wgpu::DeviceDescriptor::default(), None)
            .await
            .unwrap()
    }

    #[tokio::test]
    async fn test_pipeline_builder_creates_valid_pipeline() {
        let (device, _queue) = create_test_device().await;

        let shader = device.create_shader_module(wgpu::ShaderModuleDescriptor {
            label: Some("test_shader"),
            source: wgpu::ShaderSource::Wgsl(r#"
                @vertex fn vs_main() -> @builtin(position) vec4<f32> {
                    return vec4<f32>(0.0, 0.0, 0.0, 1.0);
                }
                @fragment fn fs_main() -> @location(0) vec4<f32> {
                    return vec4<f32>(1.0, 0.0, 0.0, 1.0);
                }
            "#.into()),
        });

        let pipeline = PipelineBuilder::new(&device, "test_pipeline", &shader)
            .color_target(wgpu::ColorTargetState {
                format: wgpu::TextureFormat::Rgba8Unorm,
                blend: Some(wgpu::BlendState::REPLACE),
                write_mask: wgpu::ColorWrites::ALL,
            })
            .build();

        // 验证 pipeline 创建成功（不 panic 即成功）
        assert_eq!(pipeline.get_bind_group_layout(0).is_ok(), false); // 没有 bind group
    }

    #[test]
    fn test_tracked_buffer_auto_accounting() {
        let (device, _queue) = pollster::block_on(create_test_device());

        let before = gpu_resource_tracker::total_bytes();
        {
            let buffer = TrackedBuffer::new(&device, &wgpu::BufferDescriptor {
                label: Some("test"),
                size: 1024,
                usage: wgpu::BufferUsages::VERTEX,
                mapped_at_creation: false,
            });
            let after_create = gpu_resource_tracker::total_bytes();
            assert_eq!(after_create - before, 1024);

            // buffer 在这里 drop
        }
        let after_drop = gpu_resource_tracker::total_bytes();
        assert_eq!(after_drop, before);
    }
```

```rust
// crates/lumino-editor/src/state/tests.rs

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn test_editor_session_new_has_empty_document() {
        let session = EditorSession::new();
        assert!(session.document().is_empty());
    }

    #[test]
    fn test_insert_note_updates_document() {
        let mut session = EditorSession::new();
        let note = NoteFactory::c4(0, 480);

        session.apply(DocumentEvent::NotesInserted {
            track: 0,
            notes: vec![note],
        }).unwrap();

        assert_eq!(session.document().track_notes(0).len(), 1);
    }

    #[test]
    fn test_undo_restores_previous_state() {
        let mut session = EditorSession::new();
        let note = NoteFactory::c4(0, 480);

        session.apply(DocumentEvent::NotesInserted {
            track: 0,
            notes: vec![note],
        }).unwrap();

        session.undo().unwrap();

        assert_eq!(session.document().track_notes(0).len(), 0);
    }

    #[test]
    fn test_redo_reapplies_undone_action() {
        let mut session = EditorSession::new();
        let note = NoteFactory::c4(0, 480);

        session.apply(DocumentEvent::NotesInserted {
            track: 0,
            notes: vec![note],
        }).unwrap();

        session.undo().unwrap();
        session.redo().unwrap();

        assert_eq!(session.document().track_notes(0).len(), 1);
    }

    #[test]
    fn test_history_limit_drops_old_entries() {
        let mut session = EditorSession::with_history_limit(3);

        for i in 0..5 {
            session.apply(DocumentEvent::NotesInserted {
                track: 0,
                notes: vec![NoteFactory::c4(i * 480, 480)],
            }).unwrap();
        }

        // 只能撤销 3 次
        session.undo().unwrap();
        session.undo().unwrap();
        session.undo().unwrap();

        // 第 4 次应该失败（无更多历史）
        assert!(session.undo().is_err());
    }
}
```

## 三、CI/CD 集成

### 3.1 测试工作流

```yaml
# .github/workflows/test.yml
name: Test

on: [push, pull_request]

jobs:
  unit-test:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - name: Install Rust
        uses: dtolnay/rust-toolchain@stable

      - name: Cache cargo
        uses: actions/cache@v4
        with:
          path: |
            ~/.cargo/registry
            ~/.cargo/git
            target
          key: ${{ runner.os }}-cargo-${{ hashFiles('**/Cargo.lock') }}

      - name: Run unit tests
        run: cargo test --lib --all-features

      - name: Run integration tests
        run: cargo test --test '*'

      - name: Generate coverage
        run: |
          cargo install cargo-tarpaulin
          cargo tarpaulin --out Xml --all-features

      - name: Upload coverage
        uses: codecov/codecov-action@v4
        with:
          file: ./cobertura.xml
          fail_ci_if_error: true
```

```rust
use criterion::{black_box, criterion_group, criterion_main, Criterion};

fn bench_insert_1m_notes(c: &mut Criterion) {
    c.bench_function("insert 1M notes", |b| {
        b.iter(|| {
            let mut list = ChunkedList::new();
            for i in 0..1_000_000 {

            black_box(list);
        });
    });
```
