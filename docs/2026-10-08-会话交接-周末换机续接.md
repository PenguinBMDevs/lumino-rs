## 附录：导出 A/B 基准源码（未跟踪，可复制到新机）

`crates/tools/export/tests/gpu_export_variance_bench.rs`：

```rust
//! 临时基准：GPU 离线导出耗时浮动测量（`#[ignore]`，手动运行）
//!
//! 用法：
//! `EXPORT_BENCH_MIDI=<mid> EXPORT_BENCH_SFZ=<sfz/sf2> EXPORT_BENCH_RUNS=3 \
//!  cargo test -p lumino-export --release -- --ignored gpu_export_variance_bench --nocapture`

use std::path::PathBuf;
use std::time::Instant;

use lumino_export::audio::config::{
    AudioBackendKind, AudioChannelMode, AudioInterpolation, AudioRenderConfig, ThreadMode,
};
use lumino_export::audio::render_audio;

#[test]
#[ignore = "基准：需 EXPORT_BENCH_MIDI / EXPORT_BENCH_SFZ 环境变量"]
fn gpu_export_variance_bench() {
    let midi = PathBuf::from(std::env::var("EXPORT_BENCH_MIDI").expect("EXPORT_BENCH_MIDI"));
    let sfz = PathBuf::from(std::env::var("EXPORT_BENCH_SFZ").expect("EXPORT_BENCH_SFZ"));
    let runs: usize = std::env::var("EXPORT_BENCH_RUNS")
        .ok()
        .and_then(|s| s.parse().ok())
        .unwrap_or(3);
    let out = PathBuf::from(
        std::env::var("EXPORT_BENCH_OUT")
            .unwrap_or_else(|_| "target/tmp_export_bench.wav".to_string()),
    );

    println!("MIDI: {:?}", midi);
    println!("SFZ:  {:?}", sfz);
    println!("runs: {runs}");

    for run in 0..runs {
        let mut config = AudioRenderConfig::default();
        config.midi_path = midi.clone();
        config.soundfonts = vec![sfz.clone()];
        config.output_path = out.clone();
        config.sample_rate = 48_000;
        config.channels = AudioChannelMode::Stereo;
        config.interpolation = AudioInterpolation::Linear;
        config.apply_limiter = true;
        config.channel_threading = ThreadMode::None;
        config.key_threading = ThreadMode::None;
        config.backend = AudioBackendKind::Gpu;

        let t = Instant::now();
        render_audio(&config).expect("GPU 导出失败");
        println!("RUN {run}: {:?}", t.elapsed());
    }
}
```

> 注意：`EXPORT_BENCH_OUT` 必须用**绝对路径**（集成测试的 CWD 是 crate 目录）。
